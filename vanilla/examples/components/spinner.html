    Spinner

    A loading indicator that shows activity or progress with a spin animation.

    Installation

    <link rel="stylesheet" href="css/components/spinner.css">

    Default

    <span class="spinner" role="status" aria-label="Loading"></span>
    /* Required CSS */
@import "css/tokens.css";
@import "css/components/spinner.css";

    Sizes

    <span class="spinner spinner-sm" role="status" aria-label="Loading"></span>
<span class="spinner" role="status" aria-label="Loading"></span>
<span class="spinner spinner-lg" role="status" aria-label="Loading"></span>
<span class="spinner spinner-xl" role="status" aria-label="Loading"></span>

    Colors

    <!-- Inherits current text color -->
<span class="spinner" role="status" aria-label="Loading"></span>

<!-- Primary color variant -->
<span class="spinner spinner-primary" role="status" aria-label="Loading"></span>

<!-- Muted color variant -->
<span class="spinner spinner-muted" role="status" aria-label="Loading"></span>

<!-- Custom color via inline style -->
<span class="spinner" style="color: var(--destructive);" role="status" aria-label="Loading"></span>

    With Button

    <button class="btn btn-default" disabled>
  <span class="spinner spinner-sm" role="status" aria-label="Loading"></span>
  Loading...
</button>

    With Text

    Loading content... 
    <div style="display: flex; align-items: center; gap: 0.5rem;">
  <span class="spinner" role="status" aria-label="Loading"></span>
  <span>Loading content...</span>
</div>

    Accessibility

    Loading 
    <!-- Option 1: Using aria-label -->
<span class="spinner" role="status" aria-label="Loading"></span>

<!-- Option 2: Using sr-only span -->
<span class="spinner" role="status">
  <span class="sr-only">Loading</span>
</span>

    CSS Classes Reference

    Class Description
    .spinnerBase spinner class (required)
    .spinner-smSmall size (0.75rem)
    .spinner-lgLarge size (1.5rem)
    .spinner-xlExtra large size (2rem)
    .spinner-primaryPrimary color variant
    .spinner-mutedMuted color variant
    .sr-onlyScreen reader only text (accessibility)
