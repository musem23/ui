    Label

    A text label for form controls with support for disabled and required states.

    Installation

    <link rel="stylesheet" href="css/components/label.css">

    Basic Label

    <label class="label">Email</label>
    /* Required CSS */
@import "css/tokens.css";
@import "css/components/label.css";

/* Base label styles */
.label {
  display: inline-flex;
  align-items: center;
  gap: 0.5rem;
  font-size: 0.875rem;
  line-height: 1;
  font-weight: 500;
  user-select: none;
  color: var(--foreground);
}

    Label with Input

    <div class="form-field">
  <label class="label" for="email-input">Email</label>
  <input type="email" id="email-input" class="input" placeholder="Enter your email">
</div>
    /* Required CSS */
@import "css/tokens.css";
@import "css/components/label.css";
@import "css/components/input.css";

/* Form field layout */
.form-field {
  display: flex;
  flex-direction: column;
  gap: 0.5rem;
}

    Required Label

    <label class="label label-required" for="name-input">Full Name</label>
<input type="text" id="name-input" class="input" placeholder="Enter your full name" required>
    /* Required indicator */
.label-required::after {
  content: "*";
  color: var(--destructive);
  margin-left: 0.125rem;
}

    Sizes

<label class="label">Default</label>

.label-sm {

    Muted Label

    <label class="label">Password</label>
<input type="password" class="input">
<label class="label label-muted label-sm">Must be at least 8 characters</label>
    /* Optional modifier for muted labels */
.label-muted {
  color: var(--muted-foreground);
  font-weight: 400;
}

    Disabled State

    <label class="label" data-disabled="true">Disabled Label</label>
<input type="text" class="input" disabled>
    /* Disabled state when label is within a disabled group */
.label[data-disabled="true"],
[data-disabled="true"] .label {
  pointer-events: none;
  opacity: 0.5;
}

/* Peer-disabled: When a sibling input is disabled, style the label */
.label:has(+ input:disabled),
.label:has(+ textarea:disabled),
.label:has(+ select:disabled),
input:disabled + .label,
textarea:disabled + .label,
select:disabled + .label {
  cursor: not-allowed;
  opacity: 0.5;
}
