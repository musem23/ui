    Input

    Input

    A text input field for user data entry.

    Installation

    <link rel="stylesheet" href="css/components/input.css">

    Default

    <input type="text" class="input" placeholder="Enter your name">
    /* Required CSS */
@import "css/tokens.css";
@import "css/components/input.css";

    With Label

    <label class="label" for="email">Email</label>
<input id="email" type="email" class="input" placeholder="you@example.com">

    Input Types

    <input type="text" class="input" placeholder="Text input">
<input type="password" class="input" placeholder="Password">
<input type="number" class="input" placeholder="0">
<input type="date" class="input">
<input type="file" class="input">

    States

    <!-- Default -->
<input type="text" class="input" placeholder="Default state">

<!-- Disabled -->
<input type="text" class="input" placeholder="Disabled" disabled>

<!-- With value -->
<input type="text" class="input" value="Prefilled value">

<!-- Invalid (error state) -->
<input type="text" class="input" aria-invalid="true" placeholder="Invalid input">
    /* Disabled state */
.input:disabled {
  pointer-events: none;
  cursor: not-allowed;
  opacity: 0.5;
}

/* Invalid/error state */
.input[aria-invalid="true"] {
  border-color: var(--destructive);
  box-shadow: var(--shadow-xs), 0 0 0 3px oklch(from var(--destructive) l c h / 20%);
}

    Disabled State

    <input type="text" class="input" placeholder="Disabled input" disabled>
<input type="text" class="input" value="Disabled with value" disabled>

    Error State

    Please enter a valid email address.

    <label class="label" for="error-email">Email</label>
<input id="error-email" type="email" class="input" value="invalid-email" aria-invalid="true">
<p class="error-message">Please enter a valid email address.</p>

    CSS Classes Reference

    Class / Attribute Description
    .inputBase input class (required)
    disabledDisabled state (native HTML attribute)
    aria-invalid="true"Error/invalid state styling
    type="text"Standard text input
    type="password"Password input (obscured text)
    type="email"Email input with validation
    type="number"Numeric input with spinners
    type="date"Date picker input
    type="file"File upload input
