    Usage

    HTML Structure

    <div class="carousel" data-carousel>
  <div class="carousel-content">
    <div class="carousel-track">
      <div class="carousel-item">Slide 1</div>
      <div class="carousel-item">Slide 2</div>
      <div class="carousel-item">Slide 3</div>
    </div>
  </div>
  <!-- Navigation buttons auto-generated, or add manually -->
</div>

    Data Attributes

    Attribute Default Description
    data-carousel - Required. Enables carousel.
    data-carousel-loop false Loop back to start/end.
    data-carousel-show-dots false Show dot indicators.
    data-carousel-autoplay false Auto-advance slides.
    data-carousel-autoplay-interval 5000 Autoplay interval (ms).
    data-carousel-orientation horizontal horizontal or vertical.

    JavaScript API

    // Get instance
const carousel = BrandUI.getInstance(element, 'carousel');

// Methods
carousel.next();           // Go to next slide
carousel.prev();           // Go to previous slide
carousel.goTo(2);          // Go to specific slide (0-indexed)
carousel.getCurrentIndex(); // Get current slide index
carousel.getCount();       // Get total number of slides
carousel.destroy();        // Clean up

// Events
element.addEventListener('carousel:change', (e) => {
  console.log(e.detail.currentIndex, e.detail.previousIndex);
});
