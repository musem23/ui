    Drawer

    A mobile-friendly panel that slides in from the edge of the screen with drag-to-dismiss support.

    Installation

    <link rel="stylesheet" href="css/components/drawer.css"> <script src="js/core.js"></script> <script src="js/components/drawer.js"></script>

    Bottom Drawer (Default)

    The default drawer slides up from the bottom. Drag down or swipe to dismiss.

    Edit Profile

    Make changes to your profile here. Drag down to dismiss.

    <!-- Trigger Button -->
<button
  data-drawer
  data-drawer-target="my-drawer"
  data-drawer-direction="bottom"
  aria-controls="my-drawer"
  aria-expanded="false"
>
  Open Drawer
</button>

<!-- Drawer -->
<div class="drawer-overlay" data-state="closed"></div>
<div
  id="my-drawer"
  class="drawer-content"
  data-direction="bottom"
  data-state="closed"
  role="dialog"
  aria-modal="true"
>
  <div class="drawer-handle"></div>
  <div class="drawer-header">
    <h3 class="drawer-title">Title</h3>
    <p class="drawer-description">Description</p>
  </div>
  <div class="drawer-body">
    Content here
  </div>
  <div class="drawer-footer">
    <button data-drawer-close>Close</button>
  </div>
</div>

    JavaScript API

    // Get or create instance
const drawer = BrandUI.Drawer('#my-trigger');

// Methods
drawer.open();
drawer.close();
drawer.toggle();
drawer.destroy();

// Events
element.addEventListener('drawer:open', (e) => {});
element.addEventListener('drawer:close', (e) => {});

    Options

    Option Type Default Description
    direction string 'bottom' Direction: 'top', 'bottom', 'left', 'right'
    dismissible boolean true Allow drag-to-dismiss
    closeOnOverlay boolean true Close when clicking overlay
    closeOnEscape boolean true Close when pressing Escape
