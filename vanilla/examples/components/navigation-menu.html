    Usage

    <nav class="nav-menu" data-nav-menu data-viewport="true" role="navigation">
  <ul class="nav-menu-list">
    <li class="nav-menu-item">
      <button class="nav-menu-trigger" data-value="getting-started">
        Getting Started
        <svg class="chevron-down">...</svg>
      </button>
      <div class="nav-menu-content" data-state="closed" hidden>
        <!-- Dropdown content here -->
      </div>
    </li>

    <!-- Standalone link (no dropdown) -->
    <li class="nav-menu-item">
      <a href="#" class="nav-menu-link nav-menu-trigger">
        Documentation
      </a>
    </li>
  </ul>
</nav>
    // Auto-initialized via data-nav-menu attribute

// Or manually:
const menu = BrandUI.initNavigationMenu(document.querySelector('.nav-menu'));

// API
menu.open('getting-started');  // Open specific item
menu.close();                  // Close all
    data-nav-menu          - Enables auto-initialization
data-viewport="true"   - Use shared viewport for dropdowns (default: true)
data-viewport="false"  - Dropdowns appear directly under trigger
data-delay="200"       - Delay before opening on hover (ms)
data-skip-delay="300"  - Time to skip delay after closing (ms)

    Keyboard Navigation

    Key Action
    Enter / Space Toggle dropdown
    Arrow Down Open dropdown, then navigate items
    Arrow Up Navigate items, then return to trigger
    Arrow Left/Right Navigate between menu items
    Home / End Jump to first/last menu item
    Escape Close dropdown
    Tab Move focus, closes menu when leaving

    Events

    // Listen for open/close events
menu.addEventListener('nav-menu:open', (e) => {
  console.log('Opened:', e.detail.value);
});

menu.addEventListener('nav-menu:close', () => {
  console.log('Menu closed');
});
