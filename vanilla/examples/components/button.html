    Button

    A clickable button component with multiple variants, sizes, and states.

    Installation

    <link rel="stylesheet" href="css/components/button.css">

    Variants

    <button class="btn btn-default">Default</button>
<button class="btn btn-secondary">Secondary</button>
<button class="btn btn-destructive">Destructive</button>
<button class="btn btn-outline">Outline</button>
<button class="btn btn-ghost">Ghost</button>
<button class="btn btn-link">Link</button>
    /* Required CSS */
@import "css/tokens.css";
@import "css/components/button.css";

/* Variants: btn-default, btn-secondary, btn-destructive, btn-outline, btn-ghost, btn-link */

    Sizes

    <button class="btn btn-default btn-sm">Small</button>

    CSS Classes Reference

    Class Description
    .btn Base button class (required)
    .btn-default Primary button style
    .btn-secondary Secondary button style
    .btn-destructive Destructive/danger style
    .btn-outline Outlined button style
    .btn-ghost Transparent background
    .btn-link Link-like appearance
    .btn-sm Small size
    .btn-lg Large size
    .btn-icon Square icon-only button
    .btn-icon-sm / .btn-icon-lg Small/large icon buttons
