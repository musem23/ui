    Select

    A dropdown select component for choosing from a list of options.

    Installation

    <link rel="stylesheet" href="css/components/select.css"> <script src="js/core.js"></script> <script src="js/components/select.js"></script>

    Default

    <div class="select" data-select data-placeholder="Select an option">
  <button type="button" class="select-trigger">
    <span class="select-value">Select an option</span>
  </button>
  <div class="select-content">
    <button type="button" class="select-item" data-value="option1">Option 1</button>
    <button type="button" class="select-item" data-value="option2">Option 2</button>
    <button type="button" class="select-item" data-value="option3">Option 3</button>
  </div>
</div>
    /* Required CSS */
@import "css/tokens.css";
@import "css/components/select.css";
    /* Required JS */
<script src="js/core.js"></script>
<script src="js/components/select.js"></script>

/* Auto-initialized via data-select attribute */
/* Or manually: */
const select = BrandUI.components.initSelect(element);

    Programmatic Control

    // Get select instance
const selectEl = document.getElementById('my-select');
const instance = BrandUI.getInstance(selectEl, 'select');

// Set value
instance.setValue('option1');

// Get current value
const value = instance.getValue();

// Clear selection
instance.clear();

// Disable/Enable
instance.disable();
instance.enable();

// Open/Close
instance.open();
instance.close();
instance.toggle();

// Listen for changes
selectEl.addEventListener('select:change', (e) => {
  console.log(e.detail.value, e.detail.label);
});

    Data Attributes Reference

    Attribute Description
    data-selectInitialize select component (required)
    data-placeholderPlaceholder text when no option is selected
    data-valueDefault selected value
    data-disabledDisable the select
    data-requiredMark as required for forms
    data-nameForm field name for hidden input

    CSS Classes Reference

    Class Description
    .selectContainer wrapper (required)
    .select--full-widthFull width select
    .select-triggerClickable trigger button
    .select-trigger--smSmall trigger size
    .select-valueSelected value display
    .select-contentDropdown content container
    .select-itemIndividual option item
    .select-groupGroup container for options
    .select-labelGroup label text
    .select-separatorVisual separator between items

    Keyboard Navigation

    Key Action
    Enter / SpaceOpen dropdown / Select highlighted option
    Arrow DownOpen dropdown / Navigate to next option
    Arrow UpOpen dropdown / Navigate to previous option
    HomeJump to first option
    EndJump to last option
    EscapeClose dropdown
    A-ZType-ahead search to jump to matching option
