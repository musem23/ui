    Scroll Area

    A custom styled scrollbar component that replaces native scrollbars with consistent cross-browser styling.

    Code Preview

    function initScrollArea(element) { const viewport = element.querySelector('.scroll-area-viewport'); if (!viewport) { console.warn('ScrollArea: viewport not found'); return null; } const orientation = element.dataset.orientation || 'vertical'; // Create scrollbar elements const scrollbar = createScrollbar(orientation); element.appendChild(scrollbar.track); // Set up scroll event handling viewport.addEventListener('scroll', () => { updateScrollbarPosition(scrollbar, viewport); }); // Enable thumb dragging scrollbar.thumb.addEventListener('mousedown', (e) => { handleDragStart(e, scrollbar, viewport); }); return { element, viewport, scrollTo: (options) => viewport.scrollTo(options), refresh: () => updateScrollbarPosition(scrollbar, viewport), }; }

    Usage

    <!-- Basic vertical scroll area -->
<div class="scroll-area"
     data-scroll-area
     data-scroll-area-orientation="vertical"
     style="height: 250px;">
  <div class="scroll-area-viewport">
    <!-- Your scrollable content here -->
  </div>
</div>

<!-- Horizontal scroll area -->
<div class="scroll-area"
     data-scroll-area
     data-scroll-area-orientation="horizontal">
  <div class="scroll-area-viewport">
    <div style="width: max-content;">
      <!-- Wide content here -->
    </div>
  </div>
</div>

<!-- Both directions -->
<div class="scroll-area"
     data-scroll-area
     data-scroll-area-orientation="both"
     style="height: 200px;">
  <div class="scroll-area-viewport">
    <!-- Content that overflows both ways -->
  </div>
</div>

<!-- Size variants -->
<div class="scroll-area scroll-area-sm">...</div>
<div class="scroll-area scroll-area-lg">...</div>

<!-- Always visible scrollbar -->
<div class="scroll-area scroll-area-always-visible">...</div>

    CSS Classes Reference

    Class Description
    .scroll-areaContainer wrapper (required)
    .scroll-area-viewportInner scrollable container (required)
    .scroll-area-smSmall scrollbar size
    .scroll-area-lgLarge scrollbar size
    .scroll-area-always-visibleAlways show scrollbar (not just on hover)

    Data Attributes Reference

    Attribute Description
    data-scroll-areaInitialize scroll area component (required)
    data-scroll-area-orientationSet to "vertical", "horizontal", or "both"
