    Slider

    An input for selecting a value from a range by dragging or using arrow keys.

    Installation

    <link rel="stylesheet" href="css/components/slider.css"> <script src="js/core.js"></script> <script src="js/components/slider.js"></script>

    Basic Slider

    50 
    <div class="slider" data-slider data-value="50">
  <div class="slider-track">
    <div class="slider-range"></div>
  </div>
  <div class="slider-thumb"></div>
</div>
    /* Required CSS */
@import "css/tokens.css";
@import "css/components/slider.css";

/* Structure: .slider > .slider-track > .slider-range + .slider-thumb */
    // Auto-initialized via data-slider attribute
// Or manually initialize:
const slider = BrandUI.components.Slider(element);

// Listen for changes
element.addEventListener('slider:change', (e) => {
  console.log(e.detail.value);
});

    Programmatic Control

    50 
    // Get slider instance
const slider = BrandUI.getInstance(element, 'slider');

// Get current value
console.log(slider.getValue()); // or slider.value

// Set value programmatically
slider.setValue(75);

// Disable/enable
slider.setDisabled(true);

// Listen for changes
element.addEventListener('slider:change', (e) => {
  console.log(e.detail.value);
  console.log(e.detail.min);
  console.log(e.detail.max);
});

    Data Attributes Reference

    Attribute Default Description
    data-slider-Required. Initializes the slider component.
    data-min0Minimum value of the slider range.
    data-max100Maximum value of the slider range.
    data-step1Increment/decrement step value.
    data-value0Initial value of the slider.
    data-orientationhorizontalSlider orientation: "horizontal" or "vertical".
    data-disabledfalseSet to "true" to disable the slider.

    Keyboard Navigation

    Key Action
    Arrow Right/UpIncrease value by step
    Arrow Left/DownDecrease value by step
    Shift + ArrowIncrease/decrease by 10x step
    HomeJump to minimum value
    EndJump to maximum value
    Page Up/DownLarge step (10% of range)
