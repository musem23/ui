    Programmatic Control

    Control checkbox state via JavaScript API.

    State: unchecked 
    // Get checkbox instance
const checkbox = BrandUI.getInstance(element, 'checkbox');

// Check
checkbox.check();

// Uncheck
checkbox.uncheck();

// Toggle
checkbox.toggle();

// Set indeterminate
checkbox.setIndeterminate(true);

// Get state
const { checked, indeterminate } = checkbox.getState();
