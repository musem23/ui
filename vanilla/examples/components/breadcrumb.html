    Breadcrumb

    Displays the path to the current resource using a hierarchy of links.

    Installation

    <link rel="stylesheet" href="css/components/breadcrumb.css">

    Basic

    1. Home
    3. Components
    5. Breadcrumb 
    <nav class="breadcrumb" aria-label="breadcrumb">
  <ol class="breadcrumb-list">
    <li class="breadcrumb-item">
      <a href="#" class="breadcrumb-link">Home</a>
    </li>
    <li class="breadcrumb-separator" role="presentation" aria-hidden="true">
      <svg>...</svg>
    </li>
    <li class="breadcrumb-item">
      <a href="#" class="breadcrumb-link">Components</a>
    </li>
    <li class="breadcrumb-separator" role="presentation" aria-hidden="true">
      <svg>...</svg>
    </li>
    <li class="breadcrumb-item">
      <span class="breadcrumb-page" aria-current="page">Breadcrumb</span>
    </li>
  </ol>
</nav>
    /* Required CSS */
@import "css/tokens.css";
@import "css/components/breadcrumb.css";

    CSS Classes Reference

    Class Description
    .breadcrumbBase nav container element
    .breadcrumb-listOrdered list wrapper with flexbox layout
    .breadcrumb-itemList item containing link or page text
    .breadcrumb-linkClickable link with hover state
    .breadcrumb-pageCurrent page text (non-clickable)
    .breadcrumb-separatorSeparator between items (SVG or text)
    .breadcrumb-ellipsisEllipsis indicator for collapsed items
