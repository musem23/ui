    Verified Badge

    A verification badge that can be used inline with text or positioned on avatars.

    John Doe 
    <!-- Inline with text -->
<span>John Doe <span class="verified-badge">
  <svg viewBox="0 0 22 22">
    <path class="badge-bg" d="M20.396..."></path>
    <path class="badge-check" d="M9.662..."></path>
  </svg>
</span></span>

<!-- Color variants -->
<span class="verified-badge">...</span>
<span class="verified-badge verified-badge-gold">...</span>
<span class="verified-badge verified-badge-gray">...</span>

<!-- Custom size via CSS variable -->
<span class="verified-badge" style="--verified-size: 2em;">...</span>
    /* Default uses --primary from theme */
.verified-badge {
  --verified-size: 1em;
  --verified-color: var(--primary);
}

/* Size classes */
.verified-badge-xs { --verified-size: 0.75em; }
.verified-badge-sm { --verified-size: 0.875em; }
.verified-badge-lg { --verified-size: 1.25em; }
.verified-badge-xl { --verified-size: 1.5em; }
