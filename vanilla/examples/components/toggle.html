    Toggle

    Toggle

    A two-state button that can be either on or off.

    Variants

    Default:
    Outline: 
    <!-- Default variant -->
<button class="toggle toggle-default" data-toggle aria-pressed="false">
  <svg ...></svg>
</button>

<!-- Outline variant -->
<button class="toggle toggle-outline" data-toggle aria-pressed="true">
  <svg ...></svg>
</button>

    Sizes

    Small:
    Default:
    Large: 
    <button class="toggle toggle-default toggle-sm">...</button>
<button class="toggle toggle-default">...</button>
<button class="toggle toggle-default toggle-lg">...</button>

    With Text

    <button class="toggle toggle-default" data-toggle>
  <svg ...></svg>
  Bold
</button>

    States

    Disabled: 
    <button class="toggle toggle-default" data-toggle disabled>
  <svg ...></svg>
</button>

    Interactive Demo

    Toggle state: false
    const toggle = document.getElementById('demo-toggle');

toggle.addEventListener('toggle:change', (e) => {
  console.log('Toggle state:', e.detail.pressed);
});

// Programmatic control
const instance = BrandUI.components.Toggle.getInstance(toggle);
instance.setPressed(true);
