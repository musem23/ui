    Sheet

    A slide-in panel that extends from the edge of the screen. Useful for navigation, forms, and secondary content.

    Installation

    <link rel="stylesheet" href="css/components/sheet.css"> <script src="js/core.js"></script> <script src="js/components/sheet.js"></script>

    Right Sheet (Default)

    <!-- Trigger -->
<button data-sheet="sheet-right">Open Right Sheet</button>

<!-- Sheet -->
<div class="sheet-overlay" id="sheet-right"></div>
<div class="sheet-content" data-side="right">
  <button class="btn btn-ghost btn-icon-sm sheet-close" aria-label="Close">
    <svg ...>...</svg>
  </button>
  <div class="sheet-header">
    <h3 class="sheet-title">Right Sheet</h3>
    <p class="sheet-description">Description text</p>
  </div>
  <div class="sheet-body">
    <!-- Content here -->
  </div>
  <div class="sheet-footer">
    <button data-sheet-close>Close</button>
  </div>
</div>
    /* Required CSS */
@import "css/tokens.css";
@import "css/components/sheet.css";

/* Sheet slides from right by default */
.sheet-content[data-side="right"] {
  top: 0;
  right: 0;
  bottom: 0;
  height: 100%;
  width: 75%;
  max-width: 24rem;
  border-left: 1px solid var(--border);
  transform: translateX(100%);
}

.sheet-content[data-side="right"][data-state="open"] {
  transform: translateX(0);
}
    // Auto-initialized via data-sheet attribute
// Or programmatic control:

// Open sheet
BrandUI.components.Sheet.open('sheet-right');

// Close sheet
BrandUI.components.Sheet.close('sheet-right');

// Get instance
const trigger = document.querySelector('[data-sheet="sheet-right"]');
const sheet = BrandUI.getInstance(trigger, 'sheet');
sheet.toggle();
sheet.isOpen(); // returns boolean

    Data Attributes Reference

    Attribute Element Description
    data-sheet="id"Trigger buttonLinks trigger to sheet overlay by ID
    data-side="right|left|top|bottom".sheet-contentDirection sheet slides from (default: right)
    data-state="open|closed".sheet-overlay, .sheet-contentCurrent state (set by JS)
    data-sheet-closeAny element inside sheetCloses the sheet when clicked

    CSS Classes Reference

    Class Description
    .sheet-overlayDark backdrop behind the sheet
    .sheet-contentMain sheet container (positioned fixed)
    .sheet-closeClose button (positioned absolute top-right)
    .sheet-headerHeader container with padding
    .sheet-titleTitle text styling
    .sheet-descriptionDescription text (muted color)
    .sheet-bodyScrollable content area
    .sheet-footerFooter with actions (margin-top: auto)

    JavaScript API

    // Static methods (by sheet ID)
BrandUI.components.Sheet.open('sheet-id');
BrandUI.components.Sheet.close('sheet-id');

// Instance methods
const trigger = document.querySelector('[data-sheet="sheet-id"]');
const sheet = BrandUI.getInstance(trigger, 'sheet');

sheet.open();      // Open the sheet
sheet.close();     // Close the sheet
sheet.toggle();    // Toggle open/closed state
sheet.isOpen();    // Returns boolean
sheet.destroy();   // Clean up event listeners
    // Events are emitted on the sheet-content element
const content = document.getElementById('sheet-id').nextElementSibling;

content.addEventListener('sheet:open', () => {
  console.log('Sheet opened');
});

content.addEventListener('sheet:close', () => {
  console.log('Sheet closed');
});

    Right Sheet

    This sheet slides in from the right side.

    This is the default position for sheets. Commonly used for detail panels, settings, or secondary navigation.

    Left Sheet

    This sheet slides in from the left side.

    Left sheets are commonly used for navigation menus, especially on mobile devices.

    Top Sheet

    This sheet slides in from the top.

    Top sheets are useful for notifications, alerts, or search interfaces.

    Bottom Sheet

    This sheet slides in from the bottom.

    Bottom sheets are popular on mobile for actions, selections, or additional content.
