    Switch

    A control that allows the user to toggle between on and off states.

    Installation

    <link rel="stylesheet" href="css/components/switch.css"> <script src="js/core.js"></script> <script src="js/components/switch.js"></script>

    Default

    <div class="switch-label">
  <button type="button" class="switch" data-switch id="switch-default">
    <span class="switch-thumb"></span>
  </button>
  <label for="switch-default">Airplane mode</label>
</div>
    /* Required CSS */
@import "css/tokens.css";
@import "css/components/switch.css";

.switch-label {
  display: flex;
  align-items: center;
  gap: 0.5rem;
}
    /* Auto-initialization via data-switch attribute */
<script src="js/core.js"></script>
<script src="js/components/switch.js"></script>

/* Listen for changes */
document.getElementById('switch-default').addEventListener('switch:change', (e) => {
  console.log('Switch is now:', e.detail.checked ? 'On' : 'Off');
});

    Data Attributes Reference

    Attribute Description
    data-switchInitializes the switch component (required)
    aria-checked="true"Sets the switch to checked state
    aria-checked="false"Sets the switch to unchecked state (default)
    aria-disabled="true"Disables the switch
    role="switch"ARIA role (auto-added by JS if missing)

    JavaScript API Reference

    Method / Event Description
    toggle()Toggle the switch state
    setChecked(bool)Set the checked state programmatically
    getChecked()Get the current checked state
    enable()Enable the switch
    disable()Disable the switch
    destroy()Remove event listeners and cleanup
    switch:changeEvent fired when state changes (detail: { checked: boolean })

    Programmatic Usage

    // Get switch instance
const switchEl = document.getElementById('my-switch');
const instance = BrandUI.getInstance(switchEl, 'switch');

// Toggle state
instance.toggle();

// Set specific state
instance.setChecked(true);

// Get current state
const isChecked = instance.getChecked();

// Enable/disable
instance.disable();
instance.enable();

// Listen for changes
switchEl.addEventListener('switch:change', (e) => {
  console.log('Switch is now:', e.detail.checked ? 'On' : 'Off');
});
